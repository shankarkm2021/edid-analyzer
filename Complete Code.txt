class CECFrame:
    """Represents a CEC frame"""
    
    def __init__(self, initiator: int, destination: int, opcode: int, operands: List[int] = None):
        """
        Create a CEC frame
        
        Args:
            initiator: Logical address of sender (0-15)
            destination: Logical address of recipient (0-15)
            opcode: Command opcode
            operands: Optional command parameters
        """
        self.initiator = initiator
        self.destination = destination
        self.opcode = opcode
        self.operands = operands or []
        
    def build(self) -> bytes:
        """Build the raw CEC frame bytes"""
        # Header byte: (initiator << 4) | destination
        header = ((self.initiator & 0x0F) << 4) | (self.destination & 0x0F)
        
        # Build frame
        frame = [header, self.opcode]
        frame.extend(self.operands)
        
        return bytes(frame)
    
    @classmethod
    def parse(cls, data: bytes) -> 'CECFrame':
        """Parse raw bytes into a CECFrame"""
        if len(data) < 2:
            raise ValueError("Frame too short")
            
        header = data[0]
        initiator = (header >> 4) & 0x0F
        destination = header & 0x0F
        opcode = data[1]
        operands = list(data[2:]) if len(data) > 2 else []
        
        return cls(initiator, destination, opcode, operands)
    
    def __str__(self) -> str:
        return (f"From: {CECAddress.get_name(self.initiator)} ({self.initiator}) -> "
                f"{CECAddress.get_name(self.destination)} ({self.destination}): "
                f"{CECOpcode.get_name(self.opcode)} (0x{self.opcode:02X}) "
                f"Operands: {[f'0x{b:02X}' for b in self.operands]}")


class CECCommandBuilder:
    """Build standard CEC commands"""
    
    @staticmethod
    def image_view_on(initiator: int = 0x0F, destination: int = CECAddress.TV) -> CECFrame:
        """Request TV to switch on and show source"""
        return CECFrame(initiator, destination, CECOpcode.IMAGE_VIEW_ON)
    
    @staticmethod
    def standby(initiator: int = 0x0F, destination: int = CECAddress.TV) -> CECFrame:
        """Request device to go to standby"""
        return CECFrame(initiator, destination, CECOpcode.STANDBY)
    
    @staticmethod
    def active_source(initiator: int, physical_address: Tuple[int, int, int, int]) -> CECFrame:
        """Report that this device is the active source"""
        # Physical address is 4 bytes: a.b.c.d encoded as 16 bits
        pa = (physical_address[0] << 12) | (physical_address[1] << 8) | (physical_address[2] << 4) | physical_address[3]
        operands = [(pa >> 8) & 0xFF, pa & 0xFF]
        return CECFrame(initiator, 0x0F, CECOpcode.ACTIVE_SOURCE, operands)
    
    @staticmethod
    def give_device_power_status(initiator: int, destination: int) -> CECFrame:
        """Request power status of a device"""
        return CECFrame(initiator, destination, CECOpcode.GIVE_DEVICE_POWER_STATUS)
    
    @staticmethod
    def user_control_pressed(initiator: int, destination: int, key: CECUserControl) -> CECFrame:
        """Simulate remote control key press"""
        return CECFrame(initiator, destination, CECOpcode.USER_CONTROL_PRESSED, [key])
    
    @staticmethod
    def user_control_released(initiator: int, destination: int) -> CECFrame:
        """Simulate remote control key release"""
        return CECFrame(initiator, destination, CECOpcode.USER_CONTROL_RELEASED)
    
    @staticmethod
    def set_osd_string(initiator: int, destination: int, text: str) -> CECFrame:
        """Set OSD text on display"""
        # Display text (max 13 characters)
        if len(text) > 13:
            text = text[:13]
        
        # Display control byte: 0 = display for default time
        operands = [0]
        operands.extend([ord(c) for c in text])
        
        return CECFrame(initiator, destination, CECOpcode.SET_OSD_STRING, operands)
    
    @staticmethod
    def system_audio_mode_request(initiator: int, destination: int = CECAddress.AUDIO_SYSTEM) -> CECFrame:
        """Request system audio mode"""
        return CECFrame(initiator, destination, CECOpcode.SYSTEM_AUDIO_MODE_REQUEST)
    
    @staticmethod
    def give_cec_version(initiator: int, destination: int) -> CECFrame:
        """Request CEC version supported by device"""
        return CECFrame(initiator, destination, CECOpcode.GET_CEC_VERSION)

            return None
        return frame.operands[0]
